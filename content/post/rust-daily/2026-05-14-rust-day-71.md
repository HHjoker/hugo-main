### 2. 定义 CLI 结构

使用 clap 的 derive 模式简化参数解析：

```rust
// src/cli.rs
use clap::{Parser, Subcommand};

#[derive(Parser)]
#[command(
    name = "rustodo",
    about = "一个简洁高效的任务管理器",
    version = "0.1.0"
)]
pub struct Cli {
    /// 启用详细输出
    #[arg(short, long, global = true)]
    pub verbose: bool,

    /// 指定数据文件路径
    #[arg(short, long, global = true, default_value = "rustodo.json")]
    pub file: Option<String>,

    #[command(subcommand)]
    pub command: Commands,
}

#[derive(Subcommand)]
pub enum Commands {
    /// 添加新任务
    Add {
        /// 任务内容
        #[arg(last = true)]
        content: String,

        /// 设置优先级 [low/medium/high]
        #[arg(short, long)]
        priority: Option<String>,
    },

    /// 列出所有任务
    List {
        /// 仅显示未完成的任务
        #[arg(short, long)]
        active: bool,

        /// 仅显示已完成的任务
        #[arg(short, long)]
        done: bool,

        /// 搜索关键词
        #[arg(short, long)]
        search: Option<String>,
    },

    /// 标记任务完成
    Done {
        /// 任务 ID
        id: u64,
    },

    /// 删除任务
    Delete {
        /// 任务 ID
        id: u64,
    },

    /// 编辑任务内容
    Edit {
        /// 任务 ID
        id: u64,

        /// 新内容
        #[arg(last = true)]
        content: String,
    },

    /// 设置优先级
    SetPriority {
        /// 任务 ID
        id: u64,

        /// 优先级 [low/medium/high]
        level: String,
    },
}
```

### 3. 命令路由实现

```rust
// src/main.rs
mod cli;
mod storage;
mod models;

use anyhow::Result;
use cli::Cli;
use storage::Storage;

fn main() -> Result<()> {
    let cli = Cli::parse();

    // 初始化存储
    let storage = Storage::new(cli.file.unwrap_or_else(|| "rustodo.json".to_string()))?;

    // 根据命令路由
    match &cli.command {
        cli::Commands::Add { content, priority } => {
            add_task(&storage, content, priority)?;
        }
        cli::Commands::List { active, done, search } => {
            list_tasks(&storage, *active, *done, search.as_deref())?;
        }
        cli::Commands::Done { id } => {
            mark_done(&storage, *id)?;
        }
        cli::Commands::Delete { id } => {
            delete_task(&storage, *id)?;
        }
        cli::Commands::Edit { id, content } => {
            edit_task(&storage, *id, content)?;
        }
        cli::Commands::SetPriority { id, level } => {
            set_priority(&storage, *id, level)?;
        }
    }

    Ok(())
}
```

### 4. 实现存储层

```rust
// src/storage.rs
use anyhow::{Context, Result};
use std::path::PathBuf;
use crate::models::{Task, TaskList};

#[derive(Clone)]
pub struct Storage {
    path: PathBuf,
}

impl Storage {
    pub fn new(path: String) -> Result<Self> {
        let path = PathBuf::from(path);
        Ok(Storage { path })
    }

    /// 加载任务列表
    pub fn load(&self) -> Result<TaskList> {
        if !self.path.exists() {
            return Ok(TaskList::default());
        }

        let content = std::fs::read_to_string(&self.path)
            .context(format!("读取文件失败: {:?}", self.path))?;

        let tasks: TaskList = serde_json::from_str(&content)
            .context("JSON 解析失败")?;

        Ok(tasks)
    }

    /// 保存任务列表
    pub fn save(&self, tasks: &TaskList) -> Result<()> {
        let content = serde_json::to_string_pretty(tasks)
            .context("序列化失败")?;

        std::fs::write(&self.path, content)
            .context(format!("写入文件失败: {:?}", self.path))?;

        Ok(())
    }
}
```

### 5. 实现基础命令函数

```rust
// src/commands.rs
use anyhow::{bail, Result};
use crate::models::{Priority, Task};
use crate::storage::Storage;
use std::sync::Mutex;

static TASKS: Mutex<Option<TaskList>> = Mutex::new(None);

fn with_tasks<F, R>(storage: &Storage, f: F) -> Result<R>
where
    F: FnOnce(&mut TaskList) -> Result<R>,
{
    let mut tasks = storage.load()?;
    let result = f(&mut tasks)?;
    storage.save(&tasks)?;
    Ok(result)
}

pub fn add_task(storage: &Storage, content: &str, priority: Option<&str>) -> Result<()> {
    with_tasks(storage, |tasks| {
        let priority = match priority.map(|p| p.to_lowercase()).as_deref() {
            Some("high") => Priority::High,
            Some("low") => Priority::Low,
            Some("medium") | None => Priority::Medium,
            _ => bail!("无效的优先级: {}", priority.unwrap()),
        };

        let id = tasks.next_id();
        let task = Task::new_with_priority(id, content.to_string(), priority);
        tasks.add(task);

        println!("✅ 已添加任务 #{}", id);
        Ok(())
    })
}

pub fn list_tasks(
    storage: &Storage,
    active_only: bool,
    done_only: bool,
    search: Option<&str>,
) -> Result<()> {
    let tasks = storage.load()?;

    let filtered: Vec<_> = tasks.iter()
        .filter(|t| {
            if active_only && t.completed { return false; }
            if done_only && !t.completed { return false; }
            if let Some(keyword) = search {
                if !t.content.to_lowercase().contains(&keyword.to_lowercase()) {
                    return false;
                }
            }
            true
        })
        .collect();

    if filtered.is_empty() {
        println!("暂无任务");
        return Ok(());
    }

    // 格式化输出
    println!("\n{:^6} {:<8} {:<40} {}", "ID", "状态", "内容", "优先级");
    println!("{}", "-".repeat(70));

    for task in filtered {
        let status = if task.completed { "✓" } else { "○" };
        let priority_str = match task.priority {
            Priority::High => "🔴高",
            Priority::Medium => "🟡中",
            Priority::Low => "🟢低",
        };
        println!(
            "{:^6} {:<8} {:<40} {}",
            task.id,
            status,
            if task.content.len() > 38 {
                format!("{}...", &task.content[..35])
            } else {
                task.content.clone()
            },
            priority_str
        );
    }

    println!("\n共 {} 个任务", filtered.len());
    Ok(())
}

pub fn mark_done(storage: &Storage, id: u64) -> Result<()> {
    with_tasks(storage, |tasks| {
        if let Some(task) = tasks.find_mut(id) {
            task.completed = true;
            println!("✅ 任务 #{} 已完成", id);
        } else {
            bail!("任务 #{} 不存在", id);
        }
        Ok(())
    })
}

pub fn delete_task(storage: &Storage, id: u64) -> Result<()> {
    with_tasks(storage, |tasks| {
        let position = tasks.iter().position(|t| t.id == id);
        match position {
            Some(idx) => {
                tasks.remove(idx);
                println!("🗑️ 任务 #{} 已删除", id);
            }
            None => bail!("任务 #{} 不存在", id),
        }
        Ok(())
    })
}

pub fn edit_task(storage: &Storage, id: u64, content: &str) -> Result<()> {
    with_tasks(storage, |tasks| {
        if let Some(task) = tasks.find_mut(id) {
            task.content = content.to_string();
            println!("✏️ 任务 #{} 已更新", id);
        } else {
            bail!("任务 #{} 不存在", id);
        }
        Ok(())
    })
}

pub fn set_priority(storage: &Storage, id: u64, level: &str) -> Result<()> {
    with_tasks(storage, |tasks| {
        if let Some(task) = tasks.find_mut(id) {
            task.priority = match level.to_lowercase().as_str() {
                "high" => Priority::High,
                "medium" => Priority::Medium,
                "low" => Priority::Low,
                _ => bail!("无效的优先级级别: {}", level),
            };
            println!("📌 任务 #{} 优先级已设置为 {}", id, level);
        } else {
            bail!("任务 #{} 不存在", id);
        }
        Ok(())
    })
}
```

### 6. 更新模型定义

```rust
// src/models.rs
use chrono::{DateTime, Utc};
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
#[serde(rename_all = "lowercase")]
pub enum Priority {
    Low,
    Medium,
    High,
}

impl Default for Priority {
    fn default() -> Self {
        Priority::Medium
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Task {
    pub id: u64,
    pub content: String,
    pub completed: bool,
    pub priority: Priority,
    pub created_at: DateTime<Utc>,
}

impl Task {
    pub fn new(id: u64, content: String) -> Self {
        Task {
            id,
            content,
            completed: false,
            priority: Priority::default(),
            created_at: Utc::now(),
        }
    }

    pub fn new_with_priority(id: u64, content: String, priority: Priority) -> Self {
        Task {
            id,
            content,
            completed: false,
            priority,
            created_at: Utc::now(),
        }
    }
}

#[derive(Debug, Default, Serialize, Deserialize)]
pub struct TaskList {
    pub tasks: Vec<Task>,
    next_id: u64,
}

impl TaskList {
    pub fn new() -> Self {
        TaskList::default()
    }

    pub fn add(&mut self, task: Task) {
        if task.id >= self.next_id {
            self.next_id = task.id + 1;
        }
        self.tasks.push(task);
    }

    pub fn next_id(&self) -> u64 {
        self.next_id
    }

    pub fn find_mut(&mut self, id: u64) -> Option<&mut Task> {
        self.tasks.iter_mut().find(|t| t.id == id)
    }

    pub fn iter(&self) -> impl Iterator<Item = &Task> {
        self.tasks.iter()
    }

    pub fn remove(&mut self, index: usize) -> Task {
        self.tasks.remove(index)
    }
}
```

## 关键概念理解

### 1. Clap Derive 模式

```rust
#[derive(Parser)]  // 自动实现 Cli 结构体的解析
#[command(...)]    // 配置全局命令属性
pub struct Cli { ... }

#[derive(Subcommand)]  // 自动实现子命令枚举的解析
pub enum Commands { ... }
```

这种声明式的方式比手动解析参数更简洁、更类型安全。

### 2. 命令模式的应用

```rust
match &cli.command {
    Commands::Add { content, priority } => { ... }
    Commands::List { active, done, search } => { ... }
    // ...
}
```

通过枚举 + 模式匹配实现命令分发，每个命令变体携带自己的参数。

### 3. 存储抽象

将数据持久化与业务逻辑分离：
- `Storage` 负责文件的读写
- `TaskList` 管理内存中的任务集合
- 命令函数通过闭包 `with_tasks` 操作数据
